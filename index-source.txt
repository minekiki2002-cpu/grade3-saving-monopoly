    const sanitizeCode = c => (c || "").toString().toUpperCase().replace(/[^A-Z0-9]/g, "").slice(0, 6);
    function randCode() {
      // Avoid ambiguous chars 0/O/1/I
      const letters = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
      let s = "";
      for (let i = 0; i < 4; i++) s += letters[Math.floor(Math.random() * letters.length)];
      return s;

      async claimSeat(code, seatIndex, profile) {
        await runTransaction(db, async (t) => {
          const r = roomRef(code);
          const snap = await t.get(r);
          if (!snap.exists()) throw new Error("房間不存在");
          const data = snap.data();
          if (data.status !== "lobby") throw new Error("遊戲已開始，無法加入");
          const players = [...data.game.players];
          // Release any seat this user already has
          for (let i = 0; i < players.length; i++) {
            if (players[i].userId === userId && i !== seatIndex) {
              players[i] = { ...players[i], claimed: false, userId: null };
            }
          }
          const target = players[seatIndex];
          if (target.claimed && target.userId !== userId) throw new Error("該位置已被佔用");
          players[seatIndex] = { ...target, ...profile, claimed: true, userId };
          t.update(r, { "game.players": players, updatedAt: serverTimestamp() });
        });
      },

          const data = snap.data();
          const players = [...data.game.players];
          if (players[seatIndex].userId !== userId) throw new Error("這不是你的位置");
          players[seatIndex] = { ...players[seatIndex], ...updates };
          t.update(r, { "game.players": players, updatedAt: serverTimestamp() });
        });
      },
      async releaseSeat(code, seatIndex) {
        await runTransaction(db, async (t) => {
          const r = roomRef(code);
          const snap = await t.get(r);
          if (!snap.exists()) return;
          const data = snap.data();

          players[seatIndex] = {
            ...players[seatIndex],
            claimed: false, userId: null
          };
          t.update(r, { "game.players": players, updatedAt: serverTimestamp() });
        });
      },
      async startGame(code) {
        await runTransaction(db, async (t) => {
          const r = roomRef(code);
          const snap = await t.get(r);
          if (!snap.exists()) throw new Error("房間不存在");
          const data = snap.data();
          if (data.hostId !== userId) throw new Error("只有房主可以開始遊戲");
          const claimed = data.game.players.filter(p => p.claimed);
          if (claimed.length < 2) throw new Error("至少需要 2 位玩家");
          // Find first claimed seat as starting current
          const firstClaimed = data.game.players.findIndex(p => p.claimed);
          t.update(r, {
            status: "playing",
            "game.current": firstClaimed >= 0 ? firstClaimed : 0,
            "game.startedAt": serverTimestamp(),
            updatedAt: serverTimestamp()
          });
        });
      },
      async resetRoom(code, initialGame) {
        await runTransaction(db, async (t) => {
          const r = roomRef(code);
          const snap = await t.get(r);
          if (!snap.exists()) throw new Error("房間不存在");
          if (snap.data().hostId !== userId) throw new Error("只有房主可以重置");
          t.update(r, { status: "lobby", game: initialGame, updatedAt: serverTimestamp() });
        });
      },
      async patchGame(code, patch) {
        // generic update for game state. Use carefully.
        const updates = {};
        for (const k in patch) updates[`game.${k}`] = patch[k];
        updates.updatedAt = serverTimestamp();
        await updateDoc(roomRef(code), updates);
      },
      async tx(code, mutate) {
        // Run arbitrary game-state transformation as a transaction
        await runTransaction(db, async (t) => {
          const r = roomRef(code);
          const snap = await t.get(r);
          if (!snap.exists()) throw new Error("房間不存在");
          const next = mutate(snap.data(), userId);
          if (next) {
            t.update(r, { ...next, updatedAt: serverTimestamp() });
          }
        });
      },
    };

    function movePlayerInGame(players, id, delta) {
      if (!delta) return players;
      return players.map(p => p.id === id ? { ...p, position: (p.position + delta + tiles.length * 4) % tiles.length } : p);

    function applyOutcomeToCurrent(game, sourceTitle, option) {
      const cur = game.current;
      let players = game.players.map(p => {
        if (p.id !== cur) return p;
        const baseEffect = Number(option.effect || 0);
        const shieldBlocks = p.shield && baseEffect < 0;
        const finalEffect = shieldBlocks ? 0 : baseEffect;
        const interestBonus = option.interestBonus ? Math.floor((p.saved || 0) / 500) * 50 : 0;
        const nextCash = option.setCashToZero ? 0 : (p.cash || 0) + finalEffect + interestBonus;
        const savedDelta = Number(option.savedDelta || 0);
        const nextSaved = Math.max(0, (p.saved || 0) + savedDelta);
        const ev = {
          title: sourceTitle,
          effect: finalEffect + interestBonus,
          savedDelta,
          kind: option.kind || (finalEffect >= 0 ? "收入" : "支出"),
          note: [
            option.note || "",
            shieldBlocks ? "鼓勵護盾啟動，這次罰款被抵消。" : "",
            interestBonus > 0 ? `儲蓄利息加發 ${money(interestBonus)}。` : "",
            savedDelta > 0 ? `另外直接存入目標帳戶 ${money(savedDelta)}。` : "",
          ].filter(Boolean).join(" "),
          afterCash: nextCash,
        };
        return {
          ...p,
          cash: nextCash,
          saved: nextSaved,
          skipTurns: Math.max(0, (p.skipTurns || 0) + Number(option.skipTurns || 0)),
          shield: option.shield ? true : (shieldBlocks ? false : p.shield),
          debtInterest: Math.max(p.debtInterest || 0, Number(option.debtInterest || 0)),
          ecoCount: (p.ecoCount || 0) + Number(option.ecoCount || 0),
          position: typeof option.moveTo === "number" ? option.moveTo : p.position,
          dayLog: [...(p.dayLog || []), ev],
        };
      });
      players = movePlayerInGame(players, cur, Number(option.moveDelta || 0));
      return players;
    }
    function describeOutcome(option, playerBefore, sourceTitle, correctText = "") {
      const moneyEffect = Number(option.effect || 0);
      const savedDelta = Number(option.savedDelta || 0);
      const moveDelta = Number(option.moveDelta || 0);
      const lines = [];
      if (moneyEffect > 0) lines.push(`手上現金增加 ${money(moneyEffect)}`);
      if (moneyEffect < 0) lines.push(`手上現金減少 ${money(Math.abs(moneyEffect))}`);
      if (savedDelta > 0) lines.push(`目標帳戶存入 ${money(savedDelta)}`);
      if (option.setCashToZero) lines.push("手上現金歸零");
      if (typeof option.moveTo === "number") lines.push("退回起點");
      if (moveDelta > 0) lines.push(`前進 ${moveDelta} 格`);
      if (moveDelta < 0) lines.push(`退後 ${Math.abs(moveDelta)} 格`);
      if (option.skipTurns) lines.push(`暫停 ${option.skipTurns} 回合反思`);
      if (option.shield) lines.push("得到一次鼓勵護盾");
      if (option.debtInterest) lines.push(`之後每天多付利息 ${money(option.debtInterest)}`);
      if (option.ecoCount) lines.push("環保杯紀錄 +1");
      if (option.interestBonus) lines.push(`依目前已存金額計算利息`);
      if (!lines.length) lines.push("這次金錢沒有直接增減");
      return {
        title: sourceTitle,
        playerName: playerBefore.name,
        animal: playerBefore.animal,
        color: playerBefore.color,
        label: correctText || option.label || "這次選擇",
        note: option.note || "",
        lines,
      };
    }
    function buildDaySettlement(players) {
      return players.map(p => {
        if (!p.claimed || !p.active || p.cash < 0) return p;
        let next = { ...p, dayLog: [...(p.dayLog || [])] };
        if (next.debtInterest > 0 && next.cash > 0) {
          const interest = Math.min(next.debtInterest, next.cash);
          const ev = {
            title: "借錢利息",
            effect: -interest,
            kind: "支出",
            note: `因為先借錢消費，今天要付利息 ${money(interest)}。`,
            afterCash: next.cash - interest,
          };
          next = { ...next, cash: next.cash - interest, dayLog: [...next.dayLog, ev] };
        }
        if (next.ecoCount >= 10) {
          const ev = {
            title: "環保杯集點",
            effect: 50,
            kind: "收入",
            note: "自備環保杯累積 10 次，換到免費飲料券，折算獎勵 $50。",
            afterCash: next.cash + 50,
          };
          next = { ...next, cash: next.cash + 50, ecoCount: next.ecoCount - 10, dayLog: [...next.dayLog, ev] };
        }
        const saveAmount = Math.min(next.daily, next.cash, Math.max(0, next.target - next.saved));
        if (saveAmount <= 0) return next;
        const ev = {
          title: "每日計畫存款",
          effect: 0,
          savedDelta: saveAmount,
          kind: "存款",
          note: `手上現金足夠，所以依每日計畫存入 ${money(saveAmount)} 到「${next.goal}」目標帳戶。`,
          afterCash: next.cash - saveAmount,
        };
        return { ...next, cash: next.cash - saveAmount, saved: next.saved + saveAmount, dayLog: [...next.dayLog, ev] };
      });
    }
    function advanceTurn(game) {
      const order = nextTurnPlayers(game);
      const turns = (game.turnsThisDay || 0) + 1;
      if (order.length === 0) return { ...game, sharedModal: { type: "gameover" } };
      if (turns >= order.length) {
        const settled = buildDaySettlement(game.players);
        return {
          ...game, players: settled, turnsThisDay: turns, pendingMove: null,
          sharedModal: { type: "settle", snapshot: settled.map(p => ({ ...p })) },
          message: `第 ${game.day} 天結束，請大家看結算。`,
        };
      }
      let curIdx = order.indexOf(game.current);
      let nextCur = order[(curIdx + 1) % order.length];
      let players = game.players;
      let guard = 0;
      while (players[nextCur] && players[nextCur].skipTurns > 0 && guard < order.length) {
        const skipped = players[nextCur];
        const ev = {
          title: "暫停反思",
          effect: 0,
          kind: "反思",
          note: "上一個選擇造成暫停一次，這回合先觀察別人的決定。",
          afterCash: skipped.cash,
        };
        players = players.map(p => p.id === skipped.id ? {
          ...p,
          skipTurns: Math.max(0, (p.skipTurns || 0) - 1),
          dayLog: [...(p.dayLog || []), ev],
        } : p);
        curIdx = order.indexOf(nextCur);
        nextCur = order[(curIdx + 1) % order.length];
        guard += 1;
      }
      return {
        ...game, players, current: nextCur, turnsThisDay: turns, pendingMove: null,
        message: `輪到 ${players[nextCur].name}，請擲骰。`,
      };
    }

        });
      }
      if (tile.type === "start") {
        return { ...game, players, sharedModal: { type: "choice", kind: "start", card: taskCards.start } };

      if (tile.type === "chance" || tile.type === "fate") {

      }
      if (tile.type === "flow") {
        return { ...game, players, sharedModal: { type: "flow", taskKey: tile.taskKey } };
      }
      return { ...game, players, sharedModal: { type: "choice", kind: "daily", card: taskCards[tile.taskKey] } };
    }

    function Dice({ value, rolling, onClick }) {
      const pips = {
        1: [4], 2: [0, 8], 3: [0, 4, 8], 4: [0, 2, 6, 8], 5: [0, 2, 4, 6, 8], 6: [0, 2, 3, 5, 6, 8],

      const safe = pips[value] || pips[1];
      const clickable = !!onClick;
      return <button type="button" disabled={!clickable} onClick={onClick}
                     className={`dice2d ${rolling ? "rolling" : ""} ${clickable ? "clickable" : ""}`}
                     aria-label={`骰子 ${value} 點${clickable ? "（點我擲骰）" : ""}`}>

      </button>;
    }

    function FlowTask({ task, onDone }) {
      const [items, setItems] = useState(() => [...task.steps].sort(() => Math.random() - .5));
      const [dragIndex, setDragIndex] = useState(null);
      const correct = items.every((item, i) => item === task.steps[i]);
      function move(from, to) {
        const next = [...items];
        const [picked] = next.splice(from, 1);
        next.splice(to, 0, picked);
        setItems(next);
      }
      return <div>
        <img src={task.image} className="task-image" alt="" />
        <p className="mt-2 text-lg leading-8 text-slate-700">{task.text}</p>
        <div className="mt-4 grid gap-2">
          {items.map((item, i) => <div
            key={item}
            draggable
            onDragStart={() => setDragIndex(i)}
            onDragOver={e => e.preventDefault()}
            onDrop={() => dragIndex !== null && move(dragIndex, i)}
            className="sortable-item rounded-2xl border-2 border-slate-200 bg-slate-50 p-3 font-black"
          >{i + 1}. {item}</div>)}
        </div>
        <button onClick={() => onDone(correct)} className="mt-5 w-full rounded-2xl bg-[#067ca3] px-4 py-4 text-xl font-black text-white">確認排序</button>
      </div>;
    }

    // -------------- Routing / Firebase hooks --------------
    function parseRoute() {
      const h = (window.location.hash || "").replace(/^#\/?/, "");
      const parts = h.split("/").filter(Boolean);
      if (parts[0] === "room" && parts[1]) {
        const c = ((window.firebaseGame && window.firebaseGame.sanitizeCode) || (s => (s||"").toUpperCase()))(parts[1]);
        return { name: "room", code: c };
      }
      return { name: "home" };
    }
    function navigate(path) {
      window.location.hash = path.startsWith("/") ? path : "/" + path;
    }
    function useRoute() {
      const [r, setR] = useState(parseRoute());
      useEffect(() => {
        const on = () => setR(parseRoute());
        window.addEventListener("hashchange", on);
        return () => window.removeEventListener("hashchange", on);
      }, []);
      return r;
    }
    function useFirebaseReady() {
      const [ready, setReady] = useState(Boolean(window.firebaseGame));
      useEffect(() => {
        if (window.firebaseGame) { setReady(true); return; }
        const on = () => setReady(true);
        window.addEventListener("firebase-ready", on);
        return () => window.removeEventListener("firebase-ready", on);
      }, []);
      return ready;
    }
    function useRoom(code) {
      const [s, setS] = useState({ loading: true, room: null, error: null });
      useEffect(() => {
        if (!code || !window.firebaseGame) return;
        setS({ loading: true, room: null, error: null });
        const unsub = window.firebaseGame.subscribeRoom(code,
          r => setS({ loading: false, room: r, error: null }),
          e => setS({ loading: false, room: null, error: e })
        );
        return unsub;
      }, [code]);
      return s;
    }
    function useMyPlayer(room) {
      if (!room) return { mySeat: -1, me: null };
      const uid = window.firebaseGame ? window.firebaseGame.userId : null;
      const seat = room.game.players.findIndex(p => p.userId === uid && p.claimed);
      return { mySeat: seat, me: seat >= 0 ? room.game.players[seat] : null };
    }

    function launchConfetti(accentColor) {
      if (!window.gsap) return;
      const palette = [accentColor || "#fbbf24", "#fbbf24", "#34d399", "#f472b6", "#60a5fa", "#a78bfa"];
      const N = 50;
      for (let i = 0; i < N; i++) {
        const piece = document.createElement("div");
        const isEmoji = i % 4 === 0;
        if (isEmoji) {
          piece.textContent = ["✨", "🎉", "💰", "⭐", "🎊"][i % 5];
        } else {
          piece.style.background = palette[i % palette.length];
          piece.style.width = (6 + Math.random() * 6) + "px";
          piece.style.height = (10 + Math.random() * 10) + "px";
          piece.style.borderRadius = "2px";
        }
        piece.style.cssText += `position:fixed; left:50%; top:50%; font-size:${22 + Math.random() * 10}px; z-index:80; pointer-events:none;`;
        document.body.appendChild(piece);
        const dx = (Math.random() - 0.5) * window.innerWidth * 0.95;
        const dy = (Math.random() - 0.5) * window.innerHeight * 0.7 - window.innerHeight * 0.2;
        window.gsap.fromTo(piece,
          { x: 0, y: 0, opacity: 1, rotation: 0, scale: 0.6 },
          { x: dx, y: dy + 280, opacity: 0, rotation: Math.random() * 720 - 360, scale: 1.1,
            duration: 1.7 + Math.random() * 0.6, ease: "power1.out",
            onComplete: () => piece.remove() });
      }
    }

    function RoomScreen({ code }) {
      const { loading, room, error } = useRoom(code);

        <button onClick={() => navigate("/")} className="mt-4 rounded-2xl bg-slate-700 px-5 py-2 font-black text-white">回首頁</button>
      </CenteredCard>;
      if (!room) return <CenteredCard>
        <div className="text-5xl">🔍</div>
        <div className="mt-3 font-black text-slate-600">找不到房間 {code}</div>
        <button onClick={() => navigate("/")} className="mt-4 rounded-2xl bg-rose-500 px-5 py-2 font-black text-white">回首頁</button>
      </CenteredCard>;
      if (room.status === "lobby") return <Lobby code={code} room={room} />;
      return <Game code={code} room={room} />;
    }

    // -------------- Lobby --------------
    function Lobby({ code, room }) {
      const { mySeat } = useMyPlayer(room);
      const isHost = room.hostId === window.firebaseGame.userId;
      const claimedCount = room.game.players.filter(p => p.claimed).length;
      const [busy, setBusy] = useState(false);
      const [err, setErr] = useState("");
      const [copied, setCopied] = useState(false);

      async function claim(i) {
        setBusy(true); setErr("");
        try { await window.firebaseGame.claimSeat(code, i, {}); }
        catch (e) { setErr(e.message); }
        setBusy(false);
      }
      async function release(i) {
        setBusy(true); setErr("");
        try { await window.firebaseGame.releaseSeat(code, i); }
        catch (e) { setErr(e.message); }
        setBusy(false);
      }
      async function updateMe(updates) {
        if (mySeat < 0) return;
        try { await window.firebaseGame.updateSeat(code, mySeat, updates); }
        catch (e) { setErr(e.message); }
      }
      async function onStart() {
        setBusy(true); setErr("");
        try { await window.firebaseGame.startGame(code); }
        catch (e) { setErr(e.message); }
        setBusy(false);
      }
      function copyCode() {
        try { navigator.clipboard.writeText(code); setCopied(true); setTimeout(() => setCopied(false), 1500); } catch (e) {}
      }
      function copyLink() {
        const url = window.location.origin + window.location.pathname + "#/room/" + code;
        try { navigator.clipboard.writeText(url); setCopied(true); setTimeout(() => setCopied(false), 1500); } catch (e) {}
      }

      return <main className="min-h-screen bg-gradient-to-br from-rose-100 via-amber-50 to-sky-100 p-4 md:p-8">
        <div className="mx-auto max-w-6xl">
          <header className="mb-5 flex flex-wrap items-center justify-between gap-3 rounded-3xl border-4 border-white bg-white/85 p-5 shadow-[0_18px_40px_rgba(244,114,182,.12)] backdrop-blur">
            <div>
              <p className="inline-flex rounded-full bg-amber-100 px-3 py-1 text-xs font-black text-amber-800">等待室 ｜ Lobby</p>
              <h1 className="mt-2 text-2xl font-black text-slate-800 md:text-3xl">點動物選你要當的角色 🐾</h1>
              <p className="mt-1 text-sm font-bold text-slate-500">至少 2 人才能開始</p>
            </div>
            <div className="text-right">
              <p className="text-xs font-bold text-slate-500">房號（點擊複製）</p>
              <button onClick={copyCode} className="block text-5xl font-black tracking-[0.25em] text-rose-500 transition active:scale-95">{code}</button>
              <button onClick={copyLink} className="mt-1 text-xs font-black text-sky-500 underline">複製連結</button>
              {copied && <p className="mt-1 text-xs font-black text-emerald-600">已複製 ✓</p>}
            </div>
          </header>

          <div className="grid gap-3 md:grid-cols-2 lg:grid-cols-3">
            {room.game.players.map((p, i) => (
              <SeatCard key={i} player={p}
                        mine={mySeat === i}
                        otherClaimed={p.claimed && mySeat !== i}
                        onClaim={() => claim(i)}
                        onRelease={() => release(i)}
                        onUpdate={updateMe}
                        busy={busy} />
            ))}
          </div>

          {err && <p className="mt-4 rounded-2xl bg-rose-100 p-3 text-center text-sm font-black text-rose-700">{err}</p>}

          <div className="mt-6 flex flex-wrap justify-between gap-3">
            <button onClick={() => navigate("/")} className="rounded-2xl bg-white px-5 py-3 font-black text-slate-700 shadow-md">離開回首頁</button>
            {isHost ? (
              <button disabled={busy || claimedCount < 2} onClick={onStart}
                      className="rounded-2xl bg-gradient-to-r from-emerald-400 to-teal-500 px-10 py-3 text-xl font-black text-white shadow-[0_10px_0_rgba(16,185,129,.4)] transition active:translate-y-1 active:shadow-[0_3px_0_rgba(16,185,129,.4)] disabled:bg-slate-300 disabled:shadow-none">
                {claimedCount < 2 ? `再等 ${2 - claimedCount} 人` : `開始遊戲（${claimedCount}/6）`}
              </button>
            ) : (
              <div className="self-center rounded-2xl bg-white px-5 py-2 text-sm font-black text-slate-500 shadow-md">⏳ 等房主開始（{claimedCount}/6）</div>
            )}
          </div>
        </div>
      </main>;
    }

      const prevCashRef = useRef({});
      const diceRef = useRef(null);
      const prevDiceRollingRef = useRef(false);

      // -------- GSAP: animate token movement along the board path --------
      useEffect(() => {
        if (!window.gsap) return;
        g.players.forEach(p => {
          if (!p.claimed) return;
          const el = tokenRefs.current[p.id];
          if (!el) return;
          const prev = prevPositionsRef.current[p.id];
          const curr = p.position;
          // Tile center anchor: tile is 9% wide, 13.5% tall, so its CENTER is left+4.5%, top+6.75%
          const tileCenter = idx => {
            const [tx, ty] = coordsWide[idx];
            return { left: `${tx + 4.5}%`, top: `${ty + 6.75}%` };
          };
          if (prev === undefined) {
            const c = tileCenter(curr);
            window.gsap.set(el, c);
            prevPositionsRef.current[p.id] = curr;
            return;
          }
          if (prev === curr) return;
          const steps = (curr - prev + tiles.length) % tiles.length;
          if (steps === 0 || steps > 6) {
            window.gsap.set(el, tileCenter(curr));
            prevPositionsRef.current[p.id] = curr;
            return;
          }
          window.gsap.killTweensOf(el);
          const jump = el.querySelector(".token-jump");
          const tl = window.gsap.timeline();
          for (let s = 1; s <= steps; s++) {
            const idx = (prev + s) % tiles.length;
            const c = tileCenter(idx);
            tl.to(el, { left: c.left, top: c.top, duration: 0.26, ease: "power1.inOut" });
            if (jump) {
              tl.fromTo(jump, { y: 0, scale: 1 },
                              { y: -22, scale: 1.18, duration: 0.13, ease: "power2.out" }, "<");
              tl.to(jump, { y: 0, scale: 1, duration: 0.13, ease: "bounce.out" }, ">");
            }
          }
          // Final landing wiggle
          if (jump) {
            tl.fromTo(jump, { rotation: -8 }, { rotation: 0, duration: 0.4, ease: "elastic.out(1, 0.3)" });
          }
          prevPositionsRef.current[p.id] = curr;
        });
      }, [g.players.map(p => `${p.id}:${p.position}`).join(",")]);

      // -------- GSAP: floating money text on cash change --------
      useEffect(() => {
        if (!window.gsap || !boardRef.current) return;
        g.players.forEach(p => {
          if (!p.claimed) return;
          const prev = prevCashRef.current[p.id];
          if (prev === undefined) {
            prevCashRef.current[p.id] = p.cash;
            return;
          }
          if (prev === p.cash) return;
          const delta = p.cash - prev;
          const tokenEl = tokenRefs.current[p.id];
          const boardEl = boardRef.current;
          if (tokenEl && boardEl && Math.abs(delta) > 0) {
            const float = document.createElement("div");
            float.className = `money-float ${delta > 0 ? "gain" : "loss"}`;
            float.textContent = (delta > 0 ? "+$" : "-$") + Math.abs(delta);
            // Anchor at token's left/top within board
            const tokenStyle = window.getComputedStyle(tokenEl);
            float.style.left = tokenStyle.left;
            float.style.top = tokenStyle.top;
            float.style.transform = "translate(-50%, -50%)";
            boardEl.appendChild(float);
            window.gsap.fromTo(float,
              { y: 0, opacity: 0, scale: 0.6 },
              { y: -90, opacity: 1, scale: 1.15, duration: 0.55, ease: "back.out(1.8)",
                onComplete: () => {
                  window.gsap.to(float, { opacity: 0, y: -130, duration: 0.6, ease: "power1.in",
                    onComplete: () => float.remove() });
                }
              }
            );
          }
          prevCashRef.current[p.id] = p.cash;
        });
      }, [g.players.map(p => `${p.id}:${p.cash}`).join(",")]);

      // -------- GSAP: dice "big reveal" pop when rolling completes --------
      useEffect(() => {
        if (!window.gsap || !diceRef.current) return;
        // Detect transition rolling true → false (i.e. the value just settled)
        if (prevDiceRollingRef.current && !g.rolling) {
          window.gsap.fromTo(diceRef.current,
            { scale: 1, rotation: 0 },
            { scale: 1.45, rotation: 360, duration: 0.55, ease: "back.out(1.7)",
              onComplete: () => {
                window.gsap.to(diceRef.current, { scale: 1, duration: 0.35, ease: "elastic.out(1, 0.4)" });
              }
            }
          );
        }
        prevDiceRollingRef.current = g.rolling;
      }, [g.rolling]);

      // -------- GSAP: card-draw flight animation --------
      const prevDrawingRef = useRef(null);
      useEffect(() => {
        if (!window.gsap) return;
        if (g.drawing && !prevDrawingRef.current) {
          // Fly an enlarged card from the deck into the centre of the screen.
          const isChance = g.drawing === "chance";
          const src = isChance ? "assets/card-chance-back.png" : "assets/card-fate-back.png";
          const fly = document.createElement("img");
          fly.src = src;
          fly.alt = "";
          fly.style.cssText = `position:fixed; top:50%; left:50%; width:200px; aspect-ratio:3/4; border-radius:24px; z-index:60; pointer-events:none; box-shadow:0 30px 60px rgba(0,0,0,.35); transform-origin:center;`;
          document.body.appendChild(fly);
          const startX = (isChance ? 1 : 1) * (window.innerWidth * 0.36);
          const startY = -(window.innerHeight * (isChance ? 0.20 : 0.06));
          window.gsap.fromTo(fly,
            { x: startX, y: startY, scale: 0.35, rotation: 0, opacity: 0 },
            { x: 0, y: 0, scale: 1, rotation: 360 + 180, opacity: 1, duration: 0.75, ease: "power2.out",
              onComplete: () => {
                window.gsap.to(fly, { scale: 1.08, duration: 0.18, yoyo: true, repeat: 1, ease: "sine.inOut" });
                window.gsap.to(fly, { opacity: 0, scale: 1.4, duration: 0.3, delay: 0.4, ease: "power1.in",
                  onComplete: () => fly.remove() });
              }
            }
          );
        }
        prevDrawingRef.current = g.drawing;
      }, [g.drawing]);

      // -------- Confetti burst when a player reaches their savings goal --------
      const goalReachedRef = useRef({});
      useEffect(() => {
        if (!window.gsap) return;
        g.players.forEach(p => {
          if (!p.claimed) return;
          const reached = p.saved >= p.target && p.target > 0;
          const wasReached = goalReachedRef.current[p.id] || false;
          if (reached && !wasReached) {
            launchConfetti(p.color);
          }
          goalReachedRef.current[p.id] = reached;
        });
      }, [g.players.map(p => `${p.id}:${p.saved >= p.target}`).join(",")]);


      // Load private notes
      useEffect(() => {
        if (mySeat < 0) return;
        const key = `monopoly:${code}:notes:${mySeat}`;
        setNotesValue(localStorage.getItem(key) || "");
      }, [code, mySeat]);

      function saveNotes(v) {
        setNotesValue(v);
        if (mySeat >= 0) localStorage.setItem(`monopoly:${code}:notes:${mySeat}`, v);
      }

      // Audio
      function unlockAudio() {
        if (audioReady.current) return;
        audioReady.current = true;
        const AC = window.AudioContext || window.webkitAudioContext;
        if (AC) window.gameAudio = new AC();
      }
      function beep(freq = 440, dur = .12, type = "sine", gain = .05) {
        try {
          unlockAudio();
          const ctx = window.gameAudio;
          if (!ctx) return;
          const osc = ctx.createOscillator();
          const g = ctx.createGain();
          osc.type = type;
          osc.frequency.value = freq;
          g.gain.setValueAtTime(gain, ctx.currentTime);
          g.gain.exponentialRampToValueAtTime(.001, ctx.currentTime + dur);
          osc.connect(g).connect(ctx.destination);
          osc.start();
          osc.stop(ctx.currentTime + dur);
        } catch (e) {}
      }
      function rollSound() { [180, 260, 340, 460].forEach((f, i) => setTimeout(() => beep(f, .08, "square", .035), i * 90)); }
      function cardSound() { [520, 660, 780].forEach((f, i) => setTimeout(() => beep(f, .08, "triangle", .04), i * 75)); }
      async function toggleMusic() {
        const bgm = document.getElementById("bgm");
        if (!bgm) return;
        if (!musicOn) {
          try { await bgm.play(); setMusicOn(true); } catch (e) {}
        } else {
          bgm.pause(); setMusicOn(false);
        }
      }

      // ---- Actions (only effective when isMyTurn or specified) ----
      async function rollDice() {
        if (!isMyTurn || g.rolling || g.pendingMove || g.sharedModal) return;
        unlockAudio(); rollSound();
        const value = Math.ceil(Math.random() * 6);
        try {
          // ONE atomic transaction: roll + move + clear rolling flag once written
          await window.firebaseGame.tx(code, (data) => {
            const ga = data.game;
            if (ga.rolling || ga.pendingMove || ga.sharedModal) return null;
            const cur = ga.current;
            const startPos = ga.players[cur].position;
            const passed = startPos + value >= tiles.length;
            const finalPos = (startPos + value) % tiles.length;
            const players = [...ga.players];
            players[cur] = { ...players[cur], position: finalPos };
            return {
              "game.rolling": true,
              "game.dice": value,
              "game.players": players,
              "game.pendingMove": { playerId: cur, tileIndex: finalPos, passedStart: passed },
              "game.message": `${players[cur].name} 擲出 ${value} 點，走到「${tiles[finalPos].title}」。`,
            };
          });
          // Clear rolling flag after a short local-only delay for dice shake animation
          setTimeout(async () => {
            await window.firebaseGame.tx(code, (data) => {
              if (!data.game.rolling) return null;
              return { "game.rolling": false };
            });
          }, 700);
        } catch (e) { console.error(e); }
      }

      async function openTileTask(tileIdx) {
        if (!isMyTurn || !g.pendingMove || g.pendingMove.tileIndex !== tileIdx) return;
        try {
          await window.firebaseGame.tx(code, (data) => {
            const ga = data.game;
            if (!ga.pendingMove) return null;
            const t = tiles[ga.pendingMove.tileIndex];
            const newGame = applyTileAction(ga, t, ga.pendingMove.passedStart);
            return { game: newGame };
          });
        } catch (e) { console.error(e); }
      }

      // Auto-open the tile task after the dice settles (so player doesn't need an extra click)
      useEffect(() => {
        if (!isMyTurn) return;
        if (g.rolling) return;
        if (!g.pendingMove) { autoOpenedFor.current = null; return; }
        if (g.sharedModal) return;
        const key = `${g.day}-${g.pendingMove.tileIndex}-${g.pendingMove.playerId}`;
        if (autoOpenedFor.current === key) return;
        const tileType = tiles[g.pendingMove.tileIndex].type;
        // Chance / Fate tiles need the user to click the card stack, don't auto-open
        if (tileType === "chance" || tileType === "fate") return;
        autoOpenedFor.current = key;
        const t = setTimeout(() => { openTileTask(g.pendingMove.tileIndex); }, 450);
        return () => clearTimeout(t);
      }, [g.rolling, g.pendingMove, g.sharedModal, g.day, isMyTurn]);

      async function drawCard(kind) {
        if (!isMyTurn || !g.pendingMove) return;
        const deck = kind === "chance" ? chanceCards : fateCards;
        const card = deck[Math.floor(Math.random() * deck.length)];
        cardSound();
        try {
          await window.firebaseGame.tx(code, () => ({ "game.drawing": kind }));
          setTimeout(async () => {
            await window.firebaseGame.tx(code, () => ({
              "game.drawing": null,
              "game.sharedModal": card.options
                ? { type: "choice", kind, card }
                : { type: "event", kind, card },
            }));
          }, 700);
        } catch (e) { console.error(e); }
      }

      const [actionError, setActionError] = useState("");
      function reportError(e) {
        const msg = e && (e.message || e.code) ? (e.code ? `${e.code}: ${e.message || ""}` : e.message) : String(e);
        console.error(e);
        setActionError(msg);
        setTimeout(() => setActionError(""), 5000);
      }
      async function applyChoice(option) {
        try {
          await window.firebaseGame.tx(code, (data) => {
            if (!data.game.sharedModal) return null;
            return { game: applyCardChoice(data.game, option) };
          });
        } catch (e) { reportError(e); }
      }
      async function applyEvent() {
        if (!g.sharedModal || !g.sharedModal.card) return;
        applyChoice(g.sharedModal.card);
      }
      async function finishFlow(correct) {
        try {
          await window.firebaseGame.tx(code, (data) => {
            if (!data.game.sharedModal || data.game.sharedModal.type !== "flow") return null;
            return { game: applyFlowResult(data.game, correct) };
          });
        } catch (e) { reportError(e); }
      }
      async function answerQuestion(ans) {
        try {
          await window.firebaseGame.tx(code, (data) => {
            if (!data.game.sharedModal || data.game.sharedModal.type !== "question") return null;
            return { game: applyQuestionAnswer(data.game, ans) };
          });
        } catch (e) { reportError(e); }
      }
      async function continueDay() {
        try {
          await window.firebaseGame.tx(code, (data) => {
            if (!data.game.sharedModal || data.game.sharedModal.type !== "settle") return null;
            return { game: startNextDay(data.game) };
          });
        } catch (e) { reportError(e); }
      }
      async function continueOutcome() {
        try {
          await window.firebaseGame.tx(code, (data) => {
            const modal = data.game.sharedModal;
            if (!modal || modal.type !== "outcome" || !modal.nextGame) return null;
            return { game: modal.nextGame };
          });
        } catch (e) { reportError(e); }
      }
      async function forcePassTurn() {
        if (!isHost) return;
        if (!window.confirm("確定要強制跳過目前這位玩家、換下一位？")) return;
        try {
          await window.firebaseGame.tx(code, (data) => ({
            game: advanceTurn({ ...data.game, pendingMove: null, sharedModal: null, rolling: false }),
          }));
        } catch (e) { reportError(e); }
      }
      async function resetToLobby() {
        if (!isHost) return;
        if (!window.confirm("確定要重置遊戲、回到等待室嗎？")) return;
        try {
          await window.firebaseGame.resetRoom(code, makeInitialGame());
        } catch (e) { console.error(e); }
      }

      const pendingTile = g.pendingMove ? tiles[g.pendingMove.tileIndex] : null;
      const canDrawChance = isMyTurn && g.pendingMove && pendingTile && pendingTile.type === "chance" && !g.sharedModal;
      const canDrawFate   = isMyTurn && g.pendingMove && pendingTile && pendingTile.type === "fate"   && !g.sharedModal;
      const allOut = activeClaimed.length === 0;

      return <main className="min-h-screen bg-gradient-to-br from-rose-50 via-amber-50 to-sky-50">
        <div className="mx-auto max-w-[1500px] p-2 md:p-4">
          <header className="mb-3 flex flex-wrap items-center justify-between gap-2 rounded-2xl border-[3px] border-white bg-white/85 px-4 py-2 shadow-md backdrop-blur">
            <div className="flex flex-wrap items-center gap-3">
              <span className="rounded-full bg-amber-100 px-3 py-1 text-xs font-black text-amber-800">房號 {code}</span>
              <span className="rounded-full bg-sky-100 px-3 py-1 text-sm font-black text-sky-800">第 {g.day} 天</span>
              <div className="flex items-center gap-2">
                <AnimalAvatar animal={currentPlayer.animal} color={currentPlayer.color} size="sm" />
                <span className="text-sm font-black" style={{ color: currentPlayer.color }}>
                  {currentPlayer.name}{isMyTurn && " (你)"} 的回合
                </span>
              </div>
            </div>
            <div className="flex items-center gap-2">
              <button onClick={toggleMusic} className="rounded-xl bg-sky-100 px-3 py-1.5 text-xs font-black text-sky-800">{musicOn ? "🔇 關音樂" : "🎵 播音樂"}</button>
              {isHost && <button onClick={resetToLobby} className="rounded-xl bg-slate-200 px-3 py-1.5 text-xs font-black text-slate-700">重置</button>}
              <button onClick={() => navigate("/")} className="rounded-xl bg-rose-100 px-3 py-1.5 text-xs font-black text-rose-700">離開</button>
            </div>
          </header>

          <div className="grid gap-3 xl:grid-cols-[1fr_360px]">
            <div className="rounded-3xl border-[3px] border-white bg-white/85 p-3 shadow-lg">
              {/* Dice + status bar */}
              {(() => {
                const canRoll = isMyTurn && !g.rolling && !g.pendingMove && !g.sharedModal && !allOut;
                return <div className="mb-2 flex flex-wrap items-center gap-4 rounded-2xl bg-gradient-to-r from-rose-50 via-amber-50 to-sky-50 p-3 md:gap-6 md:p-4">
                  <div className="flex flex-col items-center gap-1">
                    <div ref={diceRef} style={{ display: "inline-block" }}>
                      <Dice value={g.dice} rolling={g.rolling} onClick={canRoll ? rollDice : null} />
                    </div>
                    <p className="text-[10px] font-black text-slate-500">
                      {canRoll ? "👆 點我擲骰" : g.rolling ? "擲骰中..." : "骰子"}
                    </p>
                  </div>
                  <div className="grid min-w-[88px] place-items-center rounded-2xl border-2 border-rose-200 bg-white px-3 py-2 shadow-inner">
                    <p className="text-[10px] font-black text-slate-500">點數</p>
                    <p className="text-5xl font-black leading-none text-rose-500 tabular-nums">{g.dice}</p>
                  </div>
                  <div className="min-w-0 flex-1 px-1 text-center">
                    <p className="text-sm font-black leading-tight text-slate-700">{g.message}</p>
                    {!isMyTurn && !g.rolling && !g.pendingMove && !g.sharedModal && !allOut && (
                      <p className="mt-1 text-xs font-black text-slate-500">⏳ 等 {currentPlayer.name} 擲骰</p>
                    )}
                  </div>
                </div>;
              })()}

              <div className="game-board" ref={boardRef}>
                {tiles.map((t, i) => {
                  const [x, y] = coordsWide[i];
                  const canOpen = isMyTurn && g.pendingMove && g.pendingMove.tileIndex === i && !g.sharedModal;
                  return <button key={i}
                                 onClick={canOpen ? () => openTileTask(i) : undefined}
                                 disabled={!canOpen}
                                 className={`tile ${g.pendingMove && g.pendingMove.tileIndex === i ? "active" : ""}`}
                                 style={{ left: `${x}%`, top: `${y}%` }}
                                 aria-label={`${t.title} ${t.sub}`}>
                    <div>
                      <div className="tile-title">{t.title}</div>
                      <div className="tile-sub">{t.sub}</div>
                    </div>
                  </button>;
                })}

                {/* Token overlay: one per claimed player; positioned absolutely on board and animated via GSAP */}
                {g.players.filter(p => p.claimed).map(p => {
                  const sameTile = g.players.filter(q => q.claimed && q.position === p.position);
                  const idx = sameTile.findIndex(q => q.id === p.id);
                  const total = sameTile.length;
                  // Small fan offset so multiple tokens on one tile don't overlap
                  let dx = 0, dy = 0;
                  if (total > 1) {
                    const angle = ((idx / total) * Math.PI * 2) - Math.PI / 2;
                    dx = Math.cos(angle) * 14;
                    dy = Math.sin(angle) * 10;
                  }
                  return (
                    <div key={p.id}
                         ref={el => { if (el) tokenRefs.current[p.id] = el; }}
                         className="token-overlay">
                      <div className="token-jump" style={{ transform: `translate(${dx}px, ${dy}px)` }}>
                        <AnimalAvatar animal={p.animal} color={p.color} size="sm" />
                      </div>
                    </div>
                  );
                })}

                <button disabled={!canDrawChance}
                        onClick={() => drawCard("chance")}
                        className={`card-zone chance ${g.drawing === "chance" ? "drawing" : ""} ${canDrawChance ? "draw-ready" : ""}`}
                        aria-label="抽機會卡">
                  {canDrawChance && <span className="card-guide">點這裡翻機會卡</span>}
                  <img src="assets/card-chance-back.png" alt=""
                       className="absolute inset-0 h-full w-full rounded-2xl object-cover shadow-lg" />
                  {canDrawChance && <span className="absolute -bottom-4 left-1/2 -translate-x-1/2 rounded-full bg-rose-500 px-3 py-1 text-xs font-black text-white shadow-md">翻牌</span>}
                </button>
                <button disabled={!canDrawFate}
                        onClick={() => drawCard("fate")}
                        className={`card-zone fate ${g.drawing === "fate" ? "drawing" : ""} ${canDrawFate ? "draw-ready" : ""}`}
                        aria-label="抽命運卡">
                  {canDrawFate && <span className="card-guide">點這裡翻命運卡</span>}
                  <img src="assets/card-fate-back.png" alt=""
                       className="absolute inset-0 h-full w-full rounded-2xl object-cover shadow-lg" />
                  {canDrawFate && <span className="absolute -bottom-4 left-1/2 -translate-x-1/2 rounded-full bg-teal-500 px-3 py-1 text-xs font-black text-white shadow-md">翻牌</span>}
                </button>
              </div>

              {/* All players strip */}
              <div className="mt-3 grid grid-cols-2 gap-2 sm:grid-cols-3 lg:grid-cols-6">
                {claimedPlayers.map(p => (
                  <PlayerChip key={p.id} player={p} isCurrent={g.current === p.id} me={mySeat === p.id} isHost={p.userId === room.hostId} />
                ))}
              </div>
            </div>

            <aside className="grid content-start gap-3">
              {me ? <MyPanel me={me} isMyTurn={isMyTurn} notesValue={notesValue} onNotesChange={saveNotes} /> :
                    <SpectatorPanel />}
            </aside>
          </div>

          {g.sharedModal && (
            <SharedModal modal={g.sharedModal} g={g} isMyTurn={isMyTurn} isHost={isHost} currentPlayer={currentPlayer}
                         onApplyChoice={applyChoice}
                         onApplyEvent={applyEvent}
                         onFinishFlow={finishFlow}
                         onAnswerQuestion={answerQuestion}
                         onContinue={continueDay}
                         onContinueOutcome={continueOutcome}
                         onForcePass={forcePassTurn}
                         onReset={resetToLobby} />
          )}

          {actionError && (
            <div className="fixed bottom-4 left-1/2 z-[60] -translate-x-1/2 rounded-2xl bg-rose-600 px-5 py-3 text-sm font-black text-white shadow-2xl">
              ⚠️ 操作失敗：{actionError}
            </div>
          )}
        </div>
      </main>;
    }

    function SettleScreen({ modal, g, onContinue }) {
      const rowsRef = useRef(null);
      useEffect(() => {
        if (!window.gsap || !rowsRef.current) return;
        const items = rowsRef.current.querySelectorAll(".settle-row");
        window.gsap.fromTo(items,
          { opacity: 0, y: 30, scale: 0.92 },
          { opacity: 1, y: 0, scale: 1, duration: 0.45, ease: "back.out(1.4)", stagger: 0.12 }
        );
      }, []);
      const players = (modal.snapshot || []).filter(p => p.claimed);
      const champion = [...players].sort((a, b) => b.saved - a.saved)[0];
      return <>
        <span className="inline-flex rounded-full bg-amber-500 px-4 py-1 text-sm font-black text-white">一天過去</span>
        <h2 className="mt-3 text-3xl font-black">第 {g.day} 天結算</h2>
        <p className="mt-2 font-bold text-slate-600">每位玩家今天做了什麼？收入支出影響手上現金；一天結束後依每日計畫存入目標帳戶。</p>
        {champion && champion.saved > 0 && (
          <div className="mt-3 flex items-center gap-2 rounded-2xl bg-gradient-to-r from-amber-100 to-orange-100 p-3">
            <span className="text-3xl">👑</span>
            <div>
              <p className="text-xs font-black text-amber-700">今日儲蓄王</p>
              <p className="font-black text-amber-900">{champion.name} ─ 已存 ${champion.saved}</p>
            </div>
          </div>
        )}
        <div ref={rowsRef} className="mt-4 grid gap-3">
          {players.map(p => {
            const dayLog = p.dayLog || [];
            const income = dayLog.filter(x => x.kind === "收入").reduce((s, x) => s + x.effect, 0);
            const expense = Math.abs(dayLog.filter(x => x.kind === "支出").reduce((s, x) => s + x.effect, 0));
            const savedToday = dayLog.filter(x => x.kind === "存款" || x.savedDelta).reduce((s, x) => s + (x.savedDelta || 0), 0);
            const out = p.cash < 0;
            const reached = p.saved >= p.target && p.target > 0;
            return <div key={p.id} className={`settle-row rounded-2xl border-2 p-3 ${out ? "border-red-200 bg-red-50" : reached ? "border-amber-300 bg-amber-50" : "border-slate-100 bg-slate-50"}`}>
              <div className="flex flex-wrap items-center justify-between gap-2">
                <div className="flex items-center gap-2">
                  <AnimalAvatar animal={p.animal} color={p.color} size="sm" />
                  <b className="text-lg">{p.name}{reached && " 🏆"}</b>
                </div>
                <span className={`rounded-full px-3 py-1 text-sm font-black ${out ? "bg-red-600 text-white" : reached ? "bg-amber-500 text-white" : "bg-emerald-100 text-emerald-900"}`}>
                  {out ? "負債，淘汰出局" : reached ? `達成目標！$${p.saved}` : `已存 $${p.saved}`}
                </span>
              </div>
              <div className="mt-2 text-sm font-bold text-slate-600">
                收入 ${income} ｜ 支出 ${expense} ｜ 今日存入 ${savedToday} ｜ 手上現金 ${p.cash} ｜ 目標：{p.goal}
              </div>
              <ul className="mt-2 space-y-1 text-sm leading-6">
                {dayLog.length ? dayLog.map((e, i) => {
                  const parts = [];
                  if (e.effect > 0) parts.push(`手上現金 +${money(e.effect)}`);
                  if (e.effect < 0) parts.push(`手上現金 ${money(e.effect)}`);
                  if (e.savedDelta > 0) parts.push(`目標帳戶 +${money(e.savedDelta)}`);
                  if (!parts.length) parts.push("金錢沒有直接增減");
                  return <li key={i}>• {e.title}：{parts.join("，")}。{e.note}</li>;
                }) : <li>• 今天沒有記錄到事件。</li>}
              </ul>
            </div>;
          })}
        </div>
        <button onClick={onContinue} className="mt-5 w-full rounded-2xl bg-gradient-to-r from-slate-700 to-slate-900 px-4 py-4 text-xl font-black text-white shadow-[0_8px_0_rgba(15,23,42,.3)] active:translate-y-1 active:shadow-[0_2px_0_rgba(15,23,42,.3)]">
          🌅 開始下一天
        </button>
      </>;
    }

    function GameOverScreen({ g, isHost, onReset }) {
      const survivors = g.players.filter(p => p.claimed && p.saved > 0);
      const winners = [...survivors].sort((a, b) => b.saved - a.saved);
      const top = winners[0];
      const headRef = useRef(null);
      useEffect(() => {
        if (!window.gsap || !headRef.current) return;
        window.gsap.fromTo(headRef.current,
          { scale: 0.7, opacity: 0, rotation: -8 },
          { scale: 1, opacity: 1, rotation: 0, duration: 0.6, ease: "elastic.out(1, 0.5)" });
        if (top) {
          setTimeout(() => launchConfetti(top.color), 250);
          setTimeout(() => launchConfetti(top.color), 1100);
        }
      }, []);
      return <div className="text-center">
        <div ref={headRef}>
          <p className="text-7xl">{top ? "🏆" : "💸"}</p>
          <span className="mt-3 inline-flex rounded-full bg-rose-500 px-4 py-1 text-sm font-black text-white">遊戲結束</span>
          <h2 className="mt-3 text-3xl font-black">{top ? `${top.name} 是儲蓄冠軍！` : "所有玩家都出局了"}</h2>
        </div>
        {top && (
          <p className="mt-2 text-lg font-black text-amber-700">最終存款 ${top.saved} ✦ 目標 {top.goal}</p>
        )}
        <div className="mt-4 grid gap-2 text-left">
          {winners.map((p, i) => (
            <div key={p.id} className="flex items-center gap-2 rounded-2xl border-2 border-slate-100 bg-slate-50 p-3">
              <span className="w-6 text-center font-black text-slate-500">{i === 0 ? "🥇" : i === 1 ? "🥈" : i === 2 ? "🥉" : `#${i+1}`}</span>
              <AnimalAvatar animal={p.animal} color={p.color} size="sm" />
              <div className="flex-1">
                <p className="font-black text-slate-800">{p.name}</p>
                <p className="text-xs font-bold text-slate-500">目標：{p.goal}</p>
              </div>
              <span className="font-black text-emerald-700">${p.saved}</span>
            </div>
          ))}
        </div>
        {isHost && (
          <button onClick={onReset}
                  className="mt-5 w-full rounded-2xl bg-gradient-to-r from-emerald-400 to-teal-500 px-4 py-4 text-xl font-black text-white shadow-[0_8px_0_rgba(16,185,129,.4)] active:translate-y-1 active:shadow-[0_2px_0_rgba(16,185,129,.4)]">
            🔄 重新練習儲蓄和消費好習慣
          </button>
        )}
        {!isHost && (
          <div className="mt-5 grid gap-2">
            <p className="text-sm font-black text-slate-500">等房主按「重新練習儲蓄和消費好習慣」回到等待室</p>
            <button onClick={() => navigate("/")}
                    className="w-full rounded-2xl bg-slate-200 px-4 py-3 text-base font-black text-slate-700">
              回首頁重新加入
            </button>
          </div>
        )}
      </div>;
    }

    function PlayerChip({ player, isCurrent, me, isHost }) {
      const out = !player.active || player.cash < 0;
      const progress = clamp((player.saved / Math.max(1, player.target)) * 100, 0, 100);
      const reached = player.saved >= player.target && player.target > 0;
      return <div className={`relative flex items-center gap-2 rounded-2xl border-[3px] p-2 transition ${
        isCurrent ? "border-amber-400 bg-amber-50 shadow-md scale-[1.02]" :
        "border-white bg-white/70"
      } ${out ? "opacity-50 grayscale" : ""}`}>
        {isHost && <span className="absolute -top-2 -left-2 text-base" title="房主">👑</span>}
        {reached && <span className="absolute -top-2 -right-2 text-base" title="達標">🏆</span>}
        <AnimalAvatar animal={player.animal} color={player.color} size="md" />
        <div className="min-w-0 flex-1">
          <div className="truncate text-xs font-black text-slate-800">{player.name}{me && " ⭐"}</div>
          <div className="text-[10px] font-black text-slate-500">已存 ${player.saved}/${player.target}</div>
          <div className="mt-0.5 h-1.5 overflow-hidden rounded-full bg-slate-100">
            <div className="h-full bg-gradient-to-r from-emerald-400 to-teal-500 transition-all duration-500"
                 style={{ width: `${progress}%` }}></div>
          </div>
        </div>
      </div>;
    }

    function SharedModal({ modal, g, isMyTurn, isHost, currentPlayer, onApplyChoice, onApplyEvent, onFinishFlow, onAnswerQuestion, onContinue, onContinueOutcome, onForcePass, onReset }) {
      // Active player can always interact; the host can interact too (for solo testing / rescuing stuck students)
      const interactive = isMyTurn || isHost;
      const asHostOverride = !isMyTurn && isHost;
      const modalRef = useRef(null);
      useEffect(() => {
        if (!window.gsap || !modalRef.current) return;
        window.gsap.fromTo(modalRef.current,
          { scale: 0.6, opacity: 0, y: 30, rotation: -3 },
          { scale: 1, opacity: 1, y: 0, rotation: 0, duration: 0.55, ease: "back.out(1.6)" }
        );
      }, [modal.type, (modal.card && modal.card.title) || modal.taskKey || (modal.question && modal.question.q) || modal.type]);
      return <div className="modal-backdrop fixed inset-0 z-50 grid place-items-center p-4">
        <div ref={modalRef} className="max-h-[92vh] w-full max-w-3xl overflow-auto rounded-[30px] bg-white p-5 shadow-2xl">
          {isMyTurn && modal.type !== "settle" && modal.type !== "gameover" && modal.type !== "outcome" && (
            <p className="mb-3 flex items-center justify-center gap-2 rounded-2xl bg-gradient-to-r from-amber-300 to-rose-300 p-2 text-center text-base font-black text-white shadow-md">
              ✨ 你的回合，請做選擇 ✨
            </p>
          )}
          {!isMyTurn && modal.type !== "settle" && modal.type !== "gameover" && modal.type !== "outcome" && (
            <p className="mb-3 flex items-center justify-center gap-2 rounded-2xl bg-amber-100 p-2 text-center text-sm font-black text-amber-800">
              <AnimalAvatar animal={currentPlayer.animal} color={currentPlayer.color} size="sm" />
              ⏳ {currentPlayer.name} 正在做選擇{asHostOverride ? "（你是房主，可以代為操作）" : "..."}
            </p>
          )}
          {modal.type === "choice" && (
            <>
              <span className={`inline-flex rounded-full px-4 py-1 text-sm font-black text-white ${modal.kind === "chance" ? "bg-[#a980d8]" : "bg-[#067ca3]"}`}>
                {modal.kind === "chance" ? "機會卡" : "選擇題"}
              </span>
              <h2 className="mt-3 text-3xl font-black">{modal.card.title}</h2>
              {modal.card.image && <img src={modal.card.image} className="task-image" alt="" />}
              <p className="mt-3 text-lg leading-8 text-slate-700">{modal.card.text}</p>
              <div className="mt-5 grid gap-3">
                {modal.card.options.map((op, i) => (
                  <button key={i} disabled={!interactive} onClick={() => onApplyChoice(op)}
                          className="rounded-2xl border-2 border-slate-200 bg-slate-50 p-4 text-left font-black hover:border-amber-400 hover:bg-amber-50 disabled:cursor-not-allowed disabled:opacity-60">
                    {op.label}
                  </button>
                ))}
              </div>
            </>
          )}
          {modal.type === "event" && (
            <>
              <span className={`inline-flex rounded-full px-4 py-1 text-sm font-black text-white ${modal.kind === "chance" ? "bg-[#a980d8]" : "bg-[#5cb9b0]"}`}>
                {modal.kind === "chance" ? "機會卡" : "命運卡"}
              </span>
              <h2 className="mt-3 text-3xl font-black">{modal.card.title}</h2>
              <p className="mt-3 text-lg leading-8 text-slate-700">{modal.card.text}</p>

              <button disabled={!interactive} onClick={onApplyEvent}
                      className="mt-5 w-full rounded-2xl bg-[#5cb9b0] px-4 py-4 text-xl font-black text-white disabled:opacity-60">
                記錄這件事，揭曉結果
              </button>
            </>
          )}
          {modal.type === "flow" && (
            <>
              <span className="inline-flex rounded-full bg-[#067ca3] px-4 py-1 text-sm font-black text-white">流程排序任務</span>
              <h2 className="mt-3 text-3xl font-black">{flowTasks[modal.taskKey].title}</h2>
              <FlowTask task={flowTasks[modal.taskKey]} onDone={interactive ? onFinishFlow : null} />
              {!interactive && <p className="mt-3 text-center text-sm font-black text-slate-500">只有 {currentPlayer.name} 可以操作</p>}
            </>
          )}
          {modal.type === "question" && (
            <>
              <span className="inline-flex rounded-full bg-[#067ca3] px-4 py-1 text-sm font-black text-white">需要 / 想要 / 評估</span>
              <h2 className="mt-3 text-3xl font-black leading-tight">{modal.question.q}</h2>
              <div className="mt-5 grid grid-cols-3 gap-3">
                {["需要", "想要", "評估"].map(ans => (
                  <button key={ans} disabled={!interactive} onClick={() => onAnswerQuestion(ans)}
                          className="rounded-2xl bg-[#067ca3] px-3 py-5 text-xl font-black text-white disabled:opacity-60">
                    {ans}
                  </button>
                ))}
              </div>
            </>
          )}

            <div className="text-center">
              <div className="mx-auto grid h-20 w-20 place-items-center rounded-full border-4 bg-white shadow-lg" style={{ borderColor: modal.outcome.color }}>
                <AnimalAvatar animal={modal.outcome.animal} color={modal.outcome.color} size="md" />
              </div>
              <span className="mt-4 inline-flex rounded-full bg-amber-500 px-4 py-1 text-sm font-black text-white">選擇後果揭曉</span>
              <h2 className="mt-3 text-3xl font-black">{modal.outcome.title}</h2>
              <p className="mt-2 text-lg font-black text-slate-700">{modal.outcome.playerName} 選擇：{modal.outcome.label}</p>
              <div className="mt-5 grid gap-3 text-left">
                {modal.outcome.lines.map((line, i) => (
                  <div key={i} className="rounded-2xl border-2 border-amber-200 bg-amber-50 p-4 text-xl font-black text-amber-900">
                    ✦ {line}
                  </div>
                ))}
              </div>
              {modal.outcome.note && <p className="mt-4 rounded-2xl bg-slate-50 p-4 text-left font-bold leading-7 text-slate-700">{modal.outcome.note}</p>}
              <button disabled={!interactive} onClick={onContinueOutcome}
                      className="mt-5 w-full rounded-2xl bg-gradient-to-r from-emerald-400 to-teal-500 px-4 py-4 text-xl font-black text-white shadow-[0_8px_0_rgba(16,185,129,.35)] disabled:opacity-60">
                我知道了，繼續遊戲
              </button>
              {!interactive && <p className="mt-3 text-sm font-black text-slate-500">等 {currentPlayer.name} 或房主按下繼續</p>}
            </div>
          )}
          {modal.type === "settle" && (
            <SettleScreen modal={modal} g={g} onContinue={onContinue} />
          )}
          {modal.type === "gameover" && (
            <GameOverScreen g={g} isHost={isHost} onReset={onReset} />
          )}
          {isHost && modal.type !== "settle" && modal.type !== "gameover" && modal.type !== "outcome" && (
            <div className="mt-4 border-t border-slate-200 pt-3 text-center">
              <button onClick={onForcePass}
                      className="rounded-xl bg-slate-200 px-4 py-1.5 text-xs font-black text-slate-600 hover:bg-rose-200 hover:text-rose-700">
                🛟 房主：強制跳過此玩家
              </button>
            </div>
          )}
        </div>
      </div>;
    }
